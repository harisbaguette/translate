cmake_minimum_required(VERSION 3.22)
project(barobom LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
option(BAROBOM_FAST "Build dot-product CPU variant" OFF)
if(BAROBOM_FAST)
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
else()
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory(../../../../vendor/llama.cpp llama-build)
add_library(barobom SHARED translator.cpp)
target_link_libraries(barobom PRIVATE llama android log)
target_link_options(barobom PRIVATE "-Wl,-z,max-page-size=16384")

if(BAROBOM_FAST)
    set_target_properties(barobom PROPERTIES OUTPUT_NAME barobom_fast)
endif()
